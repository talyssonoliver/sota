```bash
# Analyze unused variables with context
echo "🔍 Analyzing unused variables..."
python3 -c "
import json
import re

# Load ruff issues
with open('investigation_reports/$(basename $ARGUMENTS .py)_current_ruff.json') as f:
    ruff_issues = json.load(f)

# Load file content for context analysis
with open('$ARGUMENTS', 'r') as f:
    content = f.read()
    lines = content.splitlines()

unused_vars = [issue for issue in ruff_issues if issue.get('code') == 'F841']

for issue in unused_vars:
    var_name = issue['message'].split(\"'\")[1] if \"'\" in issue['message'] else 'unknown'
    line_num = issue.get('location', {}).get('row', 1) - 1
    
    print(f'\\n🔍 Variable: {var_name} at line {line_num + 1}')
    
    # Get context around the variable
    if 0 <= line_num < len(lines):
        context_start = max(0, line_num - 2)
        context_end = min(len(lines), line_num + 3)
        print('  📝 Context:')
        for i in range(context_start, context_end):
            marker = '>>>' if i == line_num else '   '
            print(f'  {marker} {i+1}: {lines[i]}')
    
    # Check for common patterns
    line_content = lines[line_num] if line_num < len(lines) else ''
    
    # Side effect analysis
    has_function_call = bool(re.search(r'{}\s*=.*\(.*\)'.format(re.escape(var_name)), line_content))
    in_try_except = any('try:' in lines[i] or 'except' in lines[i] 
                       for i in range(max(0, line_num-5), min(len(lines), line_num+5)))
    has_side_effects = any(keyword in line_content.lower() 
                          for keyword in ['print', 'log', 'write', 'send', 'post', 'delete'])
    
    print(f'  🔧 Has function call: {has_function_call}')
    print(f'  🛡️ In try/except block: {in_try_except}')
    print(f'  ⚡ Has side effects: {has_side_effects}')
"
```

**Enhanced Decision Matrix**:
- 🟢 **Safe to Remove**: Simple assignment, no side effects, not in try/except
- 🟡 **Modify Usage**: Function call result that should be returned/used/logged
- 🔴 **Keep/Refactor**: Complex computation, error handling context, or incomplete implementation

##### **E402 (Import Ordering) - Safety Check**
```bash
# Check for import ordering safety
echo "📦 Checking import ordering safety..."
python3 -c "
import ast
import sys

try:
    with open('$ARGUMENTS', 'r') as f:
        content = f.read()
    
    # Parse AST to find imports
    tree = ast.parse(content)
    
    imports = []
    for node in ast.walk(tree):
        if isinstance(node, (ast.Import, ast.ImportFrom)):
            imports.append({
                'line': node.lineno,
                'type': 'import' if isinstance(node, ast.Import) else 'from_import',
                'module': node.module if hasattr(node, 'module') else getattr(node.names[0], 'name', ''),
                'in_try': False  # Will be enhanced
            })
    
    # Check for conditional imports (basic detection)
    lines = content.splitlines()
    for imp in imports:
        line_idx = imp['line'] - 1
        # Look for try/except blocks around imports
        context_before = ' '.join(lines[max(0, line_idx-3):line_idx]).lower()
        context_after = ' '.join(lines[line_idx:min(len(lines), line_idx+3)]).lower()
        
        if 'try:' in context_before or 'except' in context_after:
            imp['in_try'] = True
    
    print(f'📊 Total imports: {len(imports)}')
    conditional_imports = [imp for imp in imports if imp['in_try']]
    print(f'🔀 Conditional imports: {len(conditional_imports)}')
    
    if conditional_imports:
        print('⚠️ Warning: File has conditional imports - ordering changes need careful review')
    else:
        print('✅ No conditional imports detected - ordering changes likely safe')
        
except Exception as e:
    print(f'❌ Error analyzing imports: {e}')
"
```
